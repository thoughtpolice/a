# Paths from the assembled Wolfi rootfs that the minimos base keeps. One
# glob per line, where `*` stays within a path segment and `**` crosses
# them. cull.py adds the .so closure of every kept ELF binary, so
# libraries a binary links need no line here. An entry that matches
# nothing fails the build.
#
# Wolfi is fully usr-merged. /bin and /sbin link to usr/bin, /lib and
# /lib64 link to usr/lib, and cull.py doesn't descend through those
# links, so everything is matched under /usr/bin and /usr/lib.

# --- PID 1 and systemd ---
# /sbin/init reaches this init -> ../lib/systemd/systemd link through the
# /sbin -> usr/bin merge.
/usr/bin/init
/usr/lib/systemd/systemd
/usr/lib/systemd/systemd-*
# systemd's private libraries, found through the RPATH its binaries carry.
/usr/lib/systemd/libsystemd-*.so*
/usr/lib/systemd/*.so*
/usr/lib/systemd/system-generators/*
/usr/lib/systemd/user-generators/*
/usr/bin/systemctl
/usr/bin/journalctl
/usr/bin/busctl
/usr/bin/hostnamectl
/usr/bin/systemd-*
# systemctl's halt/poweroff/reboot/shutdown aliases.
/usr/bin/halt
/usr/bin/poweroff
/usr/bin/reboot
/usr/bin/shutdown

# --- dbus, the bus systemctl and friends talk over ---
/usr/bin/dbus-daemon
/usr/bin/dbus-send
/usr/share/dbus-1/system.conf
/usr/share/dbus-1/session.conf
/usr/share/dbus-1/system.d/**
/usr/share/dbus-1/session.d/**
/usr/share/dbus-1/system-services/**
/usr/share/dbus-1/services/**
/etc/dbus-1/**

# --- unit files ---
# The package set is curated, so the whole vendor unit tree is kept. The
# overlay masks units whose binaries the denylist drops.
/usr/lib/systemd/system/**
/usr/lib/systemd/system-preset/*
/usr/lib/systemd/user/**
/usr/lib/systemd/user-preset/*

# --- tmpfiles and sysctl defaults ---
# No sysusers.d. Accounts are baked into the overlay's /etc files, and
# the overlay masks systemd-sysusers.service as well.
/usr/lib/tmpfiles.d/*.conf
/usr/lib/sysctl.d/*.conf

# --- journal message catalog, for journalctl -x ---
/usr/lib/systemd/catalog/*

# --- the NSS modules nsswitch.conf names ---
/usr/lib/libnss_systemd.so.2
/usr/lib/libnss_myhostname.so.2
/usr/lib/libnss_files.so.2
/usr/lib/libnss_dns.so.2

# --- the dynamic loader, which no DT_NEEDED entry names ---
/usr/lib/ld-linux-x86-64.so.2

# --- libraries systemd dlopens ---
# systemd declares these in .note.dlopen ELF notes, which the DT_NEEDED
# closure can't see. The set covers mount units, journal compression,
# sandboxing and ACL lines in tmpfiles.
/usr/lib/libmount.so.*
/usr/lib/libzstd.so.*
/usr/lib/liblz4.so.*
/usr/lib/libseccomp.so.*
/usr/lib/libacl.so.*
/usr/lib/libaudit.so.*
/usr/lib/libselinux.so.*
# glibc dlopens libgcc_s to unwind in pthread_cancel.
/usr/lib/libgcc_s.so.*

# --- the merged-usr links /bin, /sbin, /lib and /lib64 ---
/lib
/lib64
/bin
/sbin
/usr/sbin
/usr/lib64

# --- terminfo ---
# Bash's readline needs an entry for the session's TERM. Wolfi installs
# ncurses-terminfo-base under /etc/terminfo, which ncurses searches.
/etc/terminfo/x/xterm-256color
/etc/terminfo/l/linux
/etc/terminfo/d/dumb

# --- the C.UTF-8 locale ---
# The boot contract exports LANG=C.UTF-8, and valkey for one refuses to
# start when setlocale() fails.
/usr/lib/locale/C.utf8/**

# --- CA certificates for services that make TLS connections ---
/etc/ssl/certs/ca-certificates.crt
/etc/ssl/cert.pem

# --- login shells ---
# The platform sshd refuses an account whose /etc/passwd shell doesn't
# exist, and exe.dev holds the HTTPS proxy until an SSH login succeeds.
/usr/bin/bash
/usr/bin/nologin
/usr/bin/dash

# --- mount and umount, which .mount units run ---
/usr/bin/mount
/usr/bin/umount

# --- chronyd ---
# No chronyc. Its control socket can set the clock, so it would be a
# privileged tool no account here could use, and the journal already
# says what chronyd is doing.
/usr/bin/chronyd
