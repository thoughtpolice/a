# Vendor sysctl files. systemd-sysctl reads /usr/lib/sysctl.d on the same
# pass as /etc/sysctl.d, so anything left here defeats the two-tier split:
# host-global keys must reach the kernel only from
# /usr/lib/minimos/sysctl-vm.conf, via minimos-harden.service, on a real VM.
#   50-coredump.conf points kernel.core_pattern at the culled
#     systemd-coredump binary and re-enables suid coredumps.
#   50-default.conf carries kernel.sysrq, kernel.core_uses_pid and
#     fs.protected_* — the netns-safe half is restated in
#     sysctl.d/60-minimos-hardening.conf, the global half in sysctl-vm.conf
#     at strictly stronger values.
#   50-pid-max.conf sets kernel.pid_max, which is host-global;
#     sysctl-vm.conf carries it instead.
/usr/lib/sysctl.d/50-coredump.conf
/usr/lib/sysctl.d/50-default.conf
/usr/lib/sysctl.d/50-pid-max.conf

# Vendor tmpfiles that provision image content from outside the image.
#   provision.conf writes /etc/hosts and /root/.ssh/authorized_keys from
#     system credentials (SMBIOS/fw_cfg/cmdline). minimos bakes those files;
#     nothing outside the image may rewrite them at boot, for the same reason
#     sysusers.d is culled.
#   static-nodes-permissions.conf chmods /dev/{fuse,net/tun,kvm,vfio/vfio,
#     vhost-*} to 0666 and references audio/kvm groups the baked /etc/group
#     does not define. Both the world-writable devices and the boot warnings
#     are unwanted; a composition that needs one of those nodes should grant
#     it deliberately via its own unit's DeviceAllow=.
#   20-systemd-shell-extra.conf links /etc/profile.d/70-systemd-shell-extra.sh
#     to a script this image does not ship.
/usr/lib/tmpfiles.d/provision.conf
/usr/lib/tmpfiles.d/static-nodes-permissions.conf
/usr/lib/tmpfiles.d/20-systemd-shell-extra.conf

# Vendor tmpfiles for daemons this image culls. Dead weight is the least
# of it:
#   systemd-network.conf names the systemd-network account, which no
#     longer exists here — `Failed to resolve user 'systemd-network'` on
#     every boot, the same defect class as the oom1 bus policy.
#   systemd-resolve.conf carries `L /etc/resolv.conf -> ../run/systemd/
#     resolve/stub-resolv.conf`. It is inert today only because `L`
#     leaves an existing file alone and minimos bakes that file; one
#     upstream modifier change (`L+`) would silently redirect DNS to a
#     stub no resolved is running to serve.
#   systemd-pstore.conf creates state for a binary the denylist drops.
#   20-systemd-ssh-generator.conf creates /run/sshd and a symlink into
#     /usr/lib/systemd/ssh_config.d for the ssh proxy and generator,
#     both of which are denied; the platform's sshd lives in /exe.dev and
#     reads none of it, so this only leaves a dangling link in /etc/ssh.
/usr/lib/tmpfiles.d/systemd-network.conf
/usr/lib/tmpfiles.d/systemd-resolve.conf
/usr/lib/tmpfiles.d/systemd-pstore.conf
/usr/lib/tmpfiles.d/20-systemd-ssh-generator.conf

# Bus policy for services this image does not have. /etc/dbus-1/** and
# /usr/share/dbus-1/** are kept wholesale, so these arrive with systemd
# even though every daemon they describe is culled (networkd, resolved,
# oomd, localed, timedated) or masked (hostnamed). Two reasons to drop
# them rather than leave them inert: oom1.conf names a `systemd-oom`
# account the baked /etc/passwd does not define, which makes dbus
# complain on every single boot ("Unknown username ... in message bus
# configuration file"), and each file is a set of send/receive rules for
# a bus name nothing can own — surface with no service behind it.
# Removing network1/resolve1 also retires the last references to the
# systemd-network and systemd-resolve accounts, which is why those (and
# systemd-timesync, now that chrony does the clock) leave /etc/passwd in
# the same change.
/usr/share/dbus-1/system.d/org.freedesktop.hostname1.conf
/usr/share/dbus-1/system.d/org.freedesktop.locale1.conf
/usr/share/dbus-1/system.d/org.freedesktop.network1.conf
/usr/share/dbus-1/system.d/org.freedesktop.oom1.conf
/usr/share/dbus-1/system.d/org.freedesktop.resolve1.conf
/usr/share/dbus-1/system.d/org.freedesktop.timedate1.conf

# chrony's vendor units, which the wholesale /usr/lib/systemd/system/**
# keep would otherwise pull in. minimos ships its own chronyd.service in
# /etc/systemd/system, which outranks these, but they are dropped rather
# than shadowed: the vendor unit ends by *undoing* its own hardening
# (`NoNewPrivileges=no`, plus AF_NETLINK back into RestrictAddressFamilies)
# so that the mailonchange directive can run sendmail — a binary this
# image does not have and a privilege transition it should never allow.
# chrony-wait.service blocks boot until the clock is synchronized, which
# is the wrong tradeoff for an appliance whose first source is local.
/usr/lib/systemd/system/chronyd.service
/usr/lib/systemd/system/chrony-wait.service
/usr/lib/systemd/system/chrony-wait.service.d/**
