# containerd, its two clients, the CNI plugins nerdctl's bridge network
# runs, and the iptables those plugins call. cull.py adds the libraries
# they link.
#
# No runc, crun or containerd-shim-runc-v2. gVisor, from its own overlay
# layer, is the only runtime that can start a container here, and the
# boot smoke checks that.

# The daemon and its clients.
/usr/bin/containerd
/usr/bin/ctr
/usr/bin/nerdctl

# The plugins nerdctl's bridge network uses (bridge, portmap, firewall,
# tuning), plus host-local for addresses and loopback. The package's
# other plugins stay out.
/usr/bin/bridge
/usr/bin/firewall
/usr/bin/host-local
/usr/bin/loopback
/usr/bin/portmap
/usr/bin/tuning

# iptables with only the nft backend. portmap and firewall run
# `iptables` and `ip6tables`, which the overlay links to this binary.
# Wolfi's own link points at the legacy backend. The exe.dev kernel has
# nf_tables built in, and minimos shuts off module loading early in
# boot, so a backend needing a module would never work.
/usr/bin/xtables-nft-multi

# iptables dlopens its match and target extensions, so the .so closure
# can't see them. portmap alone needs comment, multiport, tcp, udp, DNAT
# and MASQUERADE. Keep the generic, IPv4 and IPv6 families, and skip
# ebtables and arptables, whose programs the image lacks.
/usr/lib/xtables/libxt_*.so
/usr/lib/xtables/libipt_*.so
/usr/lib/xtables/libip6t_*.so
